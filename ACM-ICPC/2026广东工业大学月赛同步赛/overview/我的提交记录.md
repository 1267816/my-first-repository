#include <stdio.h>
int main() {
    //n为中心格坐标，盘子边长为2n-1
    //k为飞镖总数
    //中心格分数为n
    int n,k;
    scanf("%d %d",&n,&k);
    int size=2*n-1;
    int score[size][size];
    int i,j;
    for (i=0;i<size;i++) {
        for (j=0;j<size;j++) {
            score[i][j]=0;
        }
    }
    for (int cnt=0;cnt<n;cnt++) {
        for (i=cnt;i<size-cnt;i++) {
            for (j=cnt;j<size-cnt;j++) {
                score[i][j]++;
            }
        }
    }
    int x[k];
    int y[k];
    int sum=0;
    for (int t=0;t<k;t++) {
        scanf("%d %d",&x[t],&y[t]);
        sum+=score[x[t]-1][y[t]-1];
    }
    printf("%d",sum);
    return 0;
}

### 07 · C. 小猪玩音游 — 答案正确（2026-10-18 16:42:10，第 3 次）

```c
#include <stdio.h>
const double eps=1e-9;

int main(void) {
    // 第一行: N为歌曲数量，不大于200000
    // 第二行: M为游玩记录数量，不大于200000
    // 第三行: N个实数d_1, d_2, ..., d_N表示每首歌曲定数，不小于1.00不大于20.00
    // 接下来M行: s_i 表示第i次游玩的第s_i首歌曲, ACC为a_i %
    // 定数和ACC两位小数，不带百分号
    // 误差要求不大于10的-6次方
    int N,M;
    scanf("%d %d",&N,&M);

    double d[N]; //定数
    int i;
    for (i=0;i<N;i++) {
        scanf("%lf",&d[i]);
    }

    //计算每首歌的最高rating
    int song;
    double rate[N]; //存储
    for (i=0;i<N;i++) {
        rate[i]=0.00;
    }
    double acc;
    double rating; //临时计算
    for (i=0;i<M;i++) {
        scanf("%d %lf",&song,&acc);
        if (97.50-acc<eps) {
            rating=0;
        } else {
            rating=d[song-1]+2.15*((acc-97.50)/(100.90-97.50));
        }
        if (rating-rate[song-1]>eps) {
            rate[song-1]=rating;
        }
    }

    //求B30
    double b[30];
    for (i=0;i<30;i++) {
        b[i]=0.00;
    }
    int j,k;
    for (i=0;i<N;i++) {
        for (j=0;j<30;j++) {
            if (rate[i]-b[j]>eps) {
                for (k=29;k>j;k--) {
                    b[k]=b[k-1];
                }
                b[j]=rate[i];
                break;
            }
        }
    }

    double finalrating=0;
    for (i=0;i<30;i++) {
        finalrating+=b[i];
    }
    finalrating/=30;
    printf("%.10f\n",finalrating);

    return 0;
}
```
