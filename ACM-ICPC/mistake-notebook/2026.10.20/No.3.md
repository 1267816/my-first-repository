#include <stdio.h>

int main(void) {
    int t;
    scanf("%d",&t);

    int i,j,k;
    for (i=0;i<t;i++) {
        int n;
        scanf("%d",&n);
        int a[n];
        for (j=0;j<n;j++) {
            scanf("%d",&a[j]);
        }
        int b[n];
        int min=1000000001;
        for (j=0;j<n;j++) {
            for (k=0;k<n;k++) {
                if (a[k]<min) {
                    min=a[k];
                    b[j]=k+1;
                }
            }
            a[b[j]-1]=1000000001;
            min=1000000001;
        }
        for (j=0;j<n;j++) {
            printf("%d",b[j]);
            if (j==n-1) {
                printf("\n");
            } else {
                printf(" ");
            }
        }
    }

    return 0;
}
